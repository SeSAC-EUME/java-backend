public record EumeChatListDetailResponse(

// EumeChatRegisterService.java
@RequiredArgsConstructor
@Transactional(readOnly = true)
@Service
public class EumeChatRegisterService {
    private final EumeChatListRepository eumeChatListRepository;
    private final EumeChatSearchService eumeChatSearchService;

    @Transactional
    public EumeChatList createChatList(EumeUser eumeUser) {
        // 이미 존재하는지 확인
        if (eumeChatListRepository.existsByEumeUserId(eumeUser.getId())) {
            throw new EumeChatException(EumeChatErrorCode.CHAT_LIST_ALREADY_EXISTS);
        }

        EumeChatList chatList = EumeChatList.ofNewChatList(eumeUser);
        return eumeChatListRepository.save(chatList);
    }
}

// EumeChatService.java
@Slf4j
@RequiredArgsConstructor
@Transactional(readOnly = true)
@Service
public class EumeChatService {
    private final EumeChatSearchService eumeChatSearchService;
    private final EumeChatRegisterService eumeChatRegisterService;
    private final EumeUserSearchService eumeUserSearchService;
    private final WebClient webClient; // n8n 웹훅 호출용

    @Value("${n8n.webhook.eume-chat.url}")
    private String n8nWebhookUrl;

    @Transactional
    public EumeChatContentCreateResponse sendMessage(
        String userEmail,
        Long chatListId,
        EumeChatContentCreateRequest request
    ) {
        // 1. 사용자 조회
        EumeUser user = eumeUserSearchService.findByEmail(userEmail);

        // 2. 채팅 목록 조회 및 권한 확인
        EumeChatList chatList = eumeChatSearchService.findById(chatListId);
        validateChatListOwnership(chatList, user);

        // 3. 사용자 메시지 저장
        EumeChatContent userContent = eumeChatRegisterService.saveUserMessage(
            chatList, user, request.messageContent()
        );

        // 4. n8n 웹훅 호출하여 AI 응답 받기
        String eumeResponse = callN8nWebhook(request.messageContent());

        // 5. AI 응답 저장
        EumeChatContent eumeContent = eumeChatRegisterService.saveEumeMessage(
            chatList, user, eumeResponse
        );

        // 6. 채팅 목록 업데이트 시간 갱신
        chatList.updateLastChatTime();

        return EumeChatContentCreateResponse.from(userContent, eumeContent);
    }

    private void validateChatListOwnership(EumeChatList chatList, EumeUser user) {
        if (!chatList.getEumeUser().getId().equals(user.getId())) {
            throw new EumeChatException(EumeChatErrorCode.FORBIDDEN_ACCESS);
        }

// AdminService.java
@Slf4j
@RequiredArgsConstructor
@Transactional(readOnly = true)
@Service
public class AdminService {
    private final AdminSearchService adminSearchService;
    private final EumeAdminRepository adminRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtUtil jwtUtil;

    @Transactional
    public AdminLoginResponse login(AdminLoginRequest request, HttpServletResponse response) {
        // 1. 관리자 조회
        EumeAdmin admin = adminSearchService.findByAdminLoginId(request.adminLoginId());

        // 2. 계정 상태 확인
        if (!admin.isActive()) {
            throw new AdminException(AdminErrorCode.DEACTIVATED_ADMIN);
        }
        if (admin.isLocked()) {
            throw new AdminException(AdminErrorCode.ACCOUNT_LOCKED);
        }

        // 3. 비밀번호 확인
        if (!passwordEncoder.matches(request.adminPw(), admin.getAdminPw())) {
            admin.loginFail();
            adminRepository.save(admin);
            throw new AdminException(AdminErrorCode.INVALID_PASSWORD);
        }

        // 4. 로그인 성공 처리
        admin.loginSuccess();
        adminRepository.save(admin);

        // 5. JWT 토큰 생성 및 쿠키 설정
        // ... (기존 OAuth2SuccessHandler 패턴 참고)

        log.info("Admin login successful: {}", admin.getAdminLoginId());

        return AdminLoginResponse.from(admin);
    }
}

private void removeCookie(HttpServletResponse response, String name) {

## 7. 추가 고려 사항

### 페이지네이션 파라미터 검증
Controller에서 `@Min` 어노테이션으로 검증하는 것을 권장합니다:
```java
@GetMapping("/{chatListId}/contents")
public ResponseEntity<EumeChatContentListResponse> getContents(
    @PathVariable Long chatListId,
    @RequestParam(defaultValue = "0") @Min(0) int page,
    @RequestParam(defaultValue = "20") @Min(1) int size
) { ... }
```

### EumeChat 1인 1개 제한
- `EumeChatList`와 `EumeUser`는 `@ManyToOne` 관계이지만, 비즈니스 로직에서 1인 1개로 제한
- `EumeChatListRepository.existsByEumeUserId()`로 중복 생성 방지
- 필요 시 DB에 unique 제약조건 추가 고려: `@Table(uniqueConstraints = @UniqueConstraint(columnNames = "eume_user_id"))`

### UserChat 채팅방 참여자 관리
- 현재 `UserChatList`는 생성자(`eumeUser`)만 저장
- 다대다 채팅을 위해서는 별도의 참여자 테이블 필요할 수 있음
- 향후 확장 시 `UserChatParticipant` 엔티티 추가 고려

### Admin JWT 분리
- User JWT와 Admin JWT를 분리하여 관리할지 검토 필요
- 같은 JWT를 사용할 경우: 토큰에 role claim 추가
- 분리할 경우: Admin 전용 JWT 필터 및 설정 필요
